===== VOTER =====
module majority_voter #(
    parameter WIDTH = 32
)(
    input  logic [WIDTH-1:0] replica_a,
    input  logic [WIDTH-1:0] replica_b,
    input  logic [WIDTH-1:0] replica_c,

    output logic [WIDTH-1:0] voted_output,
    output logic             mismatch_detected
);

    // Bitwise majority voting.
    always_comb begin
        voted_output =
            (replica_a & replica_b) |
            (replica_a & replica_c) |
            (replica_b & replica_c);
    end

    // Detect any disagreement between replicas.
    always_comb begin
        mismatch_detected =
            (replica_a != replica_b) ||
            (replica_b != replica_c) ||
            (replica_a != replica_c);
    end

endmodule

===== TMR SUBSYSTEM =====
module astra_tmr_subsystem #(
    parameter WIDTH = 32
)(
    input  logic [2:0]        opcode,
    input  logic [WIDTH-1:0]  operand_a,
    input  logic [WIDTH-1:0]  operand_b,

    // XOR masks used only for fault-injection experiments.
    input  logic [WIDTH-1:0]  fault_mask_a,
    input  logic [WIDTH-1:0]  fault_mask_b,
    input  logic [WIDTH-1:0]  fault_mask_c,

    output logic [WIDTH-1:0]  replica_a,
    output logic [WIDTH-1:0]  replica_b,
    output logic [WIDTH-1:0]  replica_c,

    output logic [WIDTH-1:0]  voted_output,

    output logic              mismatch_detected,
    output logic              lane_a_mismatch,
    output logic              lane_b_mismatch,
    output logic              lane_c_mismatch
);

    logic [WIDTH-1:0] core_a_result;
    logic [WIDTH-1:0] core_b_result;
    logic [WIDTH-1:0] core_c_result;

    // ------------------------------------------------
    // Replica A
    // ------------------------------------------------

    replica_core #(
        .WIDTH(WIDTH)
    ) core_a (
        .opcode(opcode),
        .operand_a(operand_a),
        .operand_b(operand_b),
        .result(core_a_result)
    );

    // ------------------------------------------------
    // Replica B
    // ------------------------------------------------

    replica_core #(
        .WIDTH(WIDTH)
    ) core_b (
        .opcode(opcode),
        .operand_a(operand_a),
        .operand_b(operand_b),
        .result(core_b_result)
    );

    // ------------------------------------------------
    // Replica C
    // ------------------------------------------------

    replica_core #(
        .WIDTH(WIDTH)
    ) core_c (
        .opcode(opcode),
        .operand_a(operand_a),
        .operand_b(operand_b),
        .result(core_c_result)
    );

    // ------------------------------------------------
    // Fault injection
    //
    // XOR masks simulate corruption at replica outputs.
    // A zero mask means no injected corruption.
    // ------------------------------------------------

    assign replica_a = core_a_result ^ fault_mask_a;
    assign replica_b = core_b_result ^ fault_mask_b;
    assign replica_c = core_c_result ^ fault_mask_c;

    // ------------------------------------------------
    // Majority voter
    // ------------------------------------------------

    majority_voter #(
        .WIDTH(WIDTH)
    ) voter (
        .replica_a(replica_a),
        .replica_b(replica_b),
        .replica_c(replica_c),
        .voted_output(voted_output),
        .mismatch_detected(mismatch_detected)
    );

    // ------------------------------------------------
    // Replica mismatch indicators
    // ------------------------------------------------

    assign lane_a_mismatch = (replica_a != voted_output);
    assign lane_b_mismatch = (replica_b != voted_output);
    assign lane_c_mismatch = (replica_c != voted_output);

endmodule

===== ECC ENCODER =====
`timescale 1ns/1ps
module secded_encoder_32 (
    input  logic [31:0] data_in,
    output logic [38:0] encoded_data
);

    logic [38:1] hamming_codeword;

    integer data_index;
    integer code_position;
    integer parity_position;
    logic overall_parity;

    always_comb begin

        hamming_codeword = '0;
        encoded_data     = '0;

        data_index = 0;

        // -----------------------------------------------------
        // Insert 32 data bits into non-parity positions.
        // Parity positions: 1, 2, 4, 8, 16, 32
        // -----------------------------------------------------

        for (code_position = 1;
             code_position <= 38;
             code_position = code_position + 1) begin

            if ((code_position != 1)  &&
                (code_position != 2)  &&
                (code_position != 4)  &&
                (code_position != 8)  &&
                (code_position != 16) &&
                (code_position != 32)) begin

                hamming_codeword[code_position] =
                    data_in[data_index];

                data_index = data_index + 1;

            end

        end

        // -----------------------------------------------------
        // Generate six Hamming parity bits.
        //
        // Each parity bit checks positions whose binary position
        // includes that parity bit's corresponding bit.
        //
        // Even parity convention.
        // -----------------------------------------------------

        for (parity_position = 1;
             parity_position <= 32;
             parity_position = parity_position * 2) begin

            hamming_codeword[parity_position] = 1'b0;

            for (code_position = 1;
                 code_position <= 38;
                 code_position = code_position + 1) begin

                if ((code_position & parity_position) != 0) begin

                    hamming_codeword[parity_position] =
                        hamming_codeword[parity_position] ^
                        hamming_codeword[code_position];

                end

            end

        end

        // -----------------------------------------------------
        // Map Hamming positions 1..38 into encoded_data[37:0]
        // encoded_data[38] is the overall parity bit.
        // -----------------------------------------------------

        for (code_position = 1;
             code_position <= 38;
             code_position = code_position + 1) begin

            encoded_data[code_position - 1] =
                hamming_codeword[code_position];

        end

        // -----------------------------------------------------
        // Extended parity over the complete 38-bit Hamming
        // codeword.
        //
        // Overall parity bit is encoded_data[38].
        // -----------------------------------------------------

        overall_parity = 1'b0;

        for (code_position = 1;
             code_position <= 38;
             code_position = code_position + 1) begin

            overall_parity =
                overall_parity ^
                hamming_codeword[code_position];

        end

        encoded_data[38] = overall_parity;

    end

endmodule

===== ECC DECODER =====
`timescale 1ns/1ps
module secded_decoder_32 (
    input  logic [38:0] encoded_data,

    output logic [31:0] corrected_data,
    output logic [5:0]  syndrome,

    output logic error_detected,
    output logic error_corrected,
    output logic uncorrectable_error
);

    logic [38:1] hamming_codeword;
    logic [38:1] corrected_codeword;

    integer code_position;
    integer parity_position;
    integer data_index;

    logic [5:0] syndrome_value;
    logic parity_check;
    logic overall_parity;

    always_comb begin

        // -----------------------------------------------------
        // Initialize internal state
        // -----------------------------------------------------

        hamming_codeword    = '0;
        corrected_codeword  = '0;

        corrected_data     = '0;
        syndrome           = '0;

        error_detected     = 1'b0;
        error_corrected    = 1'b0;
        uncorrectable_error = 1'b0;

        syndrome_value     = 6'b000000;
        overall_parity    = 1'b0;
        data_index         = 0;

        // -----------------------------------------------------
        // Map encoded Hamming bits into positions 1..38
        // -----------------------------------------------------

        for (code_position = 1;
             code_position <= 38;
             code_position = code_position + 1) begin

            hamming_codeword[code_position] =
                encoded_data[code_position - 1];

        end

        corrected_codeword = hamming_codeword;

        // -----------------------------------------------------
// Calculate six individual parity-check results.
//
// Each parity check becomes one syndrome bit.
// -----------------------------------------------------

syndrome_value = 6'b000000;

        for (parity_position = 1;
             parity_position <= 32;
             parity_position = parity_position * 2) begin

            parity_check = 1'b0;

            for (code_position = 1;
                 code_position <= 38;
                 code_position = code_position + 1) begin

                if ((code_position & parity_position) != 0) begin

                    parity_check =
                        parity_check ^
                        hamming_codeword[code_position];

                end

            end

            case (parity_position)

                1:  syndrome_value[0] = parity_check;
                2:  syndrome_value[1] = parity_check;
                4:  syndrome_value[2] = parity_check;
                8:  syndrome_value[3] = parity_check;
                16: syndrome_value[4] = parity_check;
                32: syndrome_value[5] = parity_check;

                default: ;

            endcase

        end

        syndrome = syndrome_value;

        // -----------------------------------------------------
        // Overall parity check
        // -----------------------------------------------------

        overall_parity = encoded_data[38];

        for (code_position = 1;
             code_position <= 38;
             code_position = code_position + 1) begin

            overall_parity =
                overall_parity ^
                hamming_codeword[code_position];

        end

        // -----------------------------------------------------
        // SECDED decision
        // -----------------------------------------------------

        if ((syndrome_value == 0) &&
            (overall_parity == 0)) begin

            // No error detected.

            error_detected = 1'b0;
            error_corrected = 1'b0;
            uncorrectable_error = 1'b0;

        end

        else if ((syndrome_value != 0) &&
                 (overall_parity == 1)) begin

            // Odd number of errors.
            // A valid syndrome identifies a Hamming position.

            error_detected = 1'b1;

            if (syndrome_value <= 38) begin

                corrected_codeword[syndrome_value] =
                    ~corrected_codeword[syndrome_value];

                error_corrected = 1'b1;
                uncorrectable_error = 1'b0;

            end

            else begin

                // Syndrome points outside the valid codeword.

                error_corrected = 1'b0;
                uncorrectable_error = 1'b1;

            end

        end

        else if ((syndrome_value == 0) &&
                 (overall_parity == 1)) begin

            // Error in the overall parity bit.
            // Payload data itself is unaffected.

            error_detected = 1'b1;
            error_corrected = 1'b1;
            uncorrectable_error = 1'b0;

        end

        else begin

            // Nonzero syndrome with even overall parity.
            // Detects a double-bit error under the SECDED model.

            error_detected = 1'b1;
            error_corrected = 1'b0;
            uncorrectable_error = 1'b1;

        end

        // -----------------------------------------------------
        // Extract corrected data bits from codeword positions
        // -----------------------------------------------------

        data_index = 0;

        for (code_position = 1;
             code_position <= 38;
             code_position = code_position + 1) begin

            if ((code_position != 1)  &&
                (code_position != 2)  &&
                (code_position != 4)  &&
                (code_position != 8)  &&
                (code_position != 16) &&
                (code_position != 32)) begin

                corrected_data[data_index] =
                    corrected_codeword[code_position];

                data_index = data_index + 1;

            end

        end

    end

endmodule
